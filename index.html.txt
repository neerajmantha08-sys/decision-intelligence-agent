    // Decision Agent Business Logic (Fallback Engine)
    function runRuleBasedAIAgent(d) {
      const profit = d.revenue - d.expenses;
      const profitMargin = d.revenue > 0 ? ((profit / d.revenue) * 100).toFixed(1) : 0;
      const bestProductShare = d.salesUnits > 0 ? ((d.bestUnits / d.salesUnits) * 100).toFixed(1) : 0;
      const marketingRatio = d.revenue > 0 ? ((d.marketing / d.revenue) * 100).toFixed(1) : 0;

      // Score Calculation
      let score = 50;
      if (profitMargin > 25) score += 25;
      else if (profitMargin > 10) score += 15;
      else if (profitMargin < 0) score -= 30;

      if (d.inventory === 'Moderate') score += 15;
      else if (d.inventory === 'Low') score -= 10;
      else if (d.inventory === 'Excess / Overstocked') score -= 15;

      if (bestProductShare > 20) score += 10;

      score = Math.max(15, Math.min(100, Math.round(score)));

      let status = "Healthy";
      if (score < 50) status = "At Risk";
      else if (score < 70) status = "Moderate";

      // Business Insights
      const insights = [
        `Revenue is healthy compared with monthly expenses, delivering a ${profitMargin}% profit margin.`,
        `Customer activity shows steady demand with ${d.customers} active monthly customers.`,
        `Your top product (${d.bestProduct}) accounts for ${bestProductShare}% of total unit sales volume.`
      ];

      // Risks
      const risks = [];
      if (d.inventory === 'Low') {
        risks.push({
          title: "INVENTORY STOCKOUT RISK",
          priority: "High",
          reason: `Current stock level is low while top product represents ${bestProductShare}% of monthly sales volume.`
        });
      } else if (d.inventory === 'Excess / Overstocked') {
        risks.push({
          title: "CAPITAL TIED IN OVERSTOCK",
          priority: "High",
          reason: "Excess inventory is locking up operating cash flow that could be used for marketing."
        });
      } else {
        risks.push({
          title: "EXPENSE CONSTRICTION RISK",
          priority: "Medium",
          reason: `Monthly expenses consume ${(100 - profitMargin).toFixed(1)}% of total earned revenue.`
        });
      }

      // Opportunities
      const opps = [
        {
          title: "STRONG PRODUCT DEMAND",
          explanation: `The best-selling product '${d.bestProduct}' represents ${bestProductShare}% of sales. Consider increasing stock and running targeted promotions.`
        },
        {
          title: "CUSTOMER MONETIZATION",
          explanation: `With ${d.customers} active customers, introducing bundle deals can raise average order values.`
        }
      ];

      // Recommendations
      const recommendations = [
        {
          action: `Maintain Inventory Buffer for ${d.bestProduct}`,
          why: `This single product drives ${bestProductShare}% of sales volume and avoiding stockouts preserves profit.`,
          priority: "High"
        },
        {
          action: "Review Operating Expense Ratio",
          why: `Monthly expenses stand at ₹${d.expenses.toLocaleString()}. Cutting minor overhead boosts net profit margin directly.`,
          priority: "Medium"
        },
        {
          action: "Optimize Marketing Spend Allocation",
          why: `Marketing spend is currently ${marketingRatio}% of total revenue. Ensure channels target repeat customers.`,
          priority: "Low"
        }
      ];

      // Decision Summary
      const summary = {
        mostImportantFinding: `Business exhibits a strong ${profitMargin}% profit margin with ₹${profit.toLocaleString()} in net profit.`,
        biggestRisk: d.inventory === 'Low' ? "Potential stockouts on top revenue products." : "Expense growth reducing net operating margins.",
        biggestOpportunity: `Expanding sales and promotion of best-selling '${d.bestProduct}'.`,
        nextAction: `Ensure optimal inventory levels for ${d.bestProduct} while maintaining expense controls.`
      };

      return {
        id: 'analysis_' + Date.now(),
        data: d,
        metrics: { profit, profitMargin, bestProductShare, marketingRatio, score, status },
        insights,
        risks,
        opportunities: opps,
        recommendations,
        summary
      };
    }

    // Gemini API Engine
    async function runGeminiAIAgent(d) {
      try {
        const prompt = `Analyze this small business data and output strict JSON only:
          Business: ${d.name} (${d.type}), Goal: ${d.goal}
          Revenue: ${d.revenue}, Expenses: ${d.expenses}, Customers: ${d.customers}, Sales Units: ${d.salesUnits}
          Best Selling Product: ${d.bestProduct} (${d.bestUnits} units)
          Inventory Level: ${d.inventory}, Marketing Spend: ${d.marketing}
          
          Return JSON format:
          {
            "score": 78,
            "status": "Healthy",
            "insights": ["...", "..."],
            "risks": [{"title": "...", "priority": "High", "reason": "..."}],
            "opportunities": [{"title": "...", "explanation": "..."}],
            "recommendations": [{"action": "...", "why": "...", "priority": "High"}],
            "summary": {"mostImportantFinding": "...", "biggestRisk": "...", "biggestOpportunity": "...", "nextAction": "..."}
          }`;

        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${d.apiKey}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
        });

        const resData = await response.json();
        const text = resData.candidates[0].content.parts[0].text;
        const cleanedJson = JSON.parse(text.replace(/```json|```/g, '').trim());

        const profit = d.revenue - d.expenses;
        const profitMargin = d.revenue > 0 ? ((profit / d.revenue) * 100).toFixed(1) : 0;
        const bestProductShare = d.salesUnits > 0 ? ((d.bestUnits / d.salesUnits) * 100).toFixed(1) : 0;
        const marketingRatio = d.revenue > 0 ? ((d.marketing / d.revenue) * 100).toFixed(1) : 0;

        return {
          id: 'analysis_' + Date.now(),
          data: d,
          metrics: { profit, profitMargin, bestProductShare, marketingRatio, score: cleanedJson.score || 75, status: cleanedJson.status || "Healthy" },
          insights: cleanedJson.insights || [],
          risks: cleanedJson.risks || [],
          opportunities: cleanedJson.opportunities || [],
          recommendations: cleanedJson.recommendations || [],
          summary: cleanedJson.summary || {}
        };
      } catch (err) {
        console.warn("API Call Failed, falling back to rule engine", err);
        return runRuleBasedAIAgent(d);
      }
    }

    // Local Storage Reports Management
    function getSavedReports() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        return raw ? JSON.parse(raw) : [];
      } catch (e) {
        return [];
      }
    }

    function saveReportToStorage(report) {
      const list = getSavedReports();
      list.unshift(report);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
      updateSavedBadge();
    }

    function deleteSavedReport(id) {
      let list = getSavedReports();
      list = list.filter(r => r.id !== id);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
      renderSavedReports();
      updateSavedBadge();
    }

    function clearAllSavedReports() {
      if (confirm("Are you sure you want to delete all saved business reports?")) {
        localStorage.removeItem(STORAGE_KEY);
        renderSavedReports();
        updateSavedBadge();
      }
    }

    function updateSavedBadge() {
      const list = getSavedReports();
      const badge = document.getElementById('saved-badge');
      if (list.length > 0) {
        badge.innerText = list.length;
        badge.classList.remove('hidden');
      } else {
        badge.classList.add('hidden');
      }
    }

    function renderSavedReports() {
      const container = document.getElementById('saved-list-container');
      const list = getSavedReports();

      if (list.length === 0) {
        container.innerHTML = `
          <div class="col-span-full text-center py-12 bg-brandCard rounded-3xl border border-brandBorder space-y-3">
            <i data-lucide="bookmark-x" class="w-10 h-10 text-brandSecondary mx-auto"></i>
            <h3 class="text-base font-bold text-brandText">No Saved Analyses</h3>
            <p class="text-xs text-brandSecondary">All generated reports will be stored locally for future reference.</p>
          </div>
        `;
        lucide.createIcons();
        return;
      }

      container.innerHTML = list.map(item => `
        <div class="bg-brandCard p-6 rounded-3xl border border-brandBorder shadow-sm space-y-4 flex flex-col justify-between">
          <div class="space-y-1">
            <div class="flex justify-between items-start">
              <h3 class="font-bold text-base text-brandText">${item.data.name}</h3>
              <span class="px-2.5 py-0.5 rounded-full text-[10px] font-bold bg-brandGreen/10 text-brandGreen">
                Score: ${item.metrics.score}
              </span>
            </div>
            <p class="text-xs text-brandSecondary">${item.data.type} • ${new Date(item.data.timestamp).toLocaleDateString()}</p>
          </div>

          <div class="grid grid-cols-2 gap-2 text-xs py-2 border-y border-brandBorder">
            <div><span class="text-brandSecondary">Revenue:</span> <strong>₹${item.data.revenue.toLocaleString()}</strong></div>
            <div><span class="text-brandSecondary">Margin:</span> <strong>${item.metrics.profitMargin}%</strong></div>
          </div>

          <div class="flex gap-2 pt-1">
            <button onclick="loadSavedReport('${item.id}')" class="flex-1 py-2 rounded-xl bg-brandGreen text-white font-bold text-xs text-center">
              View Report
            </button>
            <button onclick="deleteSavedReport('${item.id}')" class="p-2 rounded-xl border border-brandBorder hover:bg-red-50 hover:text-red-600 transition-colors">
              <i data-lucide="trash" class="w-4 h-4"></i>
            </button>
          </div>
        </div>
      `).join('');

      lucide.createIcons();
    }

    function loadSavedReport(id) {
      const list = getSavedReports();
      const report = list.find(r => r.id === id);
      if (report) {
        currentAnalysis = report;
        renderDashboard(report);
        switchTab('dashboard');
      }
    }

    function exportReportToPDF() {
      const btn = document.getElementById('btn-export-pdf');
      const element = document.getElementById('pdf-content-area');

      // Visual feedback on export button
      const originalText = btn.innerHTML;
      btn.innerHTML = `<i data-lucide="loader-2" class="w-4 h-4 animate-spin"></i> Generating PDF...`;
      lucide.createIcons();

      const businessName = currentAnalysis ? currentAnalysis.data.name.replace(/\s+/g, '_') : 'Business';
      const opt = {
        margin:       [10, 10, 10, 10],
        filename:     `${businessName}_Decision_Report.pdf`,
        image:        { type: 'jpeg', quality: 0.98 },
        html2canvas:  { scale: 2, useCORS: true, logging: false },
        jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }
      };

      // Run html2pdf to download file directly
      if (typeof html2pdf !== 'undefined') {
        html2pdf().set(opt).from(element).save().then(() => {
          btn.innerHTML = originalText;
          lucide.createIcons();
        }).catch(err => {
          console.warn("html2pdf failed, falling back to window print", err);
          btn.innerHTML = originalText;
          lucide.createIcons();
          window.print();
        });
      } else {
        btn.innerHTML = originalText;
        lucide.createIcons();
        window.print();
      }
    }
  </script>
